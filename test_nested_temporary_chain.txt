struct Point
{
    i32 x
    i32 y

    fn d2 := () -> i32
    {
        i32 result{x * x + y * y}
        exit result
    }
}

struct Circle
{
    Point center
    i32 radius

    fn grown := (i32 dr) -> Circle
    {
        Circle result{center, radius + dr}
        exit result
    }
}

Point p{3, 4}
Circle c{p, 2}
i32 result{c.grown(1).center.d2()}
exit result
